import yfinance as yf
import numpy as np
import matplotlib.pyplot as plt
import pandas as pd


# Get 3 years of historical data for Apple stock
data = yf.download('AAPL', start='2023-01-01', end='2026-01-01')
close_prices = data['Close'].squeeze()


# Calculate the log returns
log_returns = np.log(close_prices / close_prices.shift(1))


# Calculate drift
drift = log_returns.mean()


def get_terminal_prices(number_of_simulations, volatility):


    t_intervals = 252


    # get the random element with a standard normal distribution
    random_shocks = np.random.normal(0, 1, (t_intervals, number_of_simulations))


    #Get a array ready for the price list
    price_list = np.zeros_like(random_shocks)
    price_list[0] = close_prices.iloc[-1]


    #calculate the daily growth factor matrix
    daily_multipliers = np.zeros_like(random_shocks)
    daily_multipliers = np.exp(drift + volatility * random_shocks)


    # Loop through days to recursively fill out each path
    for t in range(1, t_intervals):
        price_list[t] = price_list[t-1] * daily_multipliers[t]


    #calculate terminal prices
    end_prices = price_list[t_intervals - 1, :]
    return end_prices


def test_volatility_for_variance(volatility):


    # get a big list ready to hold the results of the simulations
    sim_results = np.zeros(100)


    # do 100 different whole simulations and calculate the variance of the terminal prices for each, adding this to sim_results
    for i in range(100):
        sim = get_terminal_prices(100, volatility)
        sim_results[i] = sim.var()


    return sim_results.mean()


x_axis = np.arange(0.04, 0.1, 0.001)
y_axis = np.zeros(len(x_axis))
for i in range(len(x_axis)):
    y_axis[i] = test_volatility_for_variance(x_axis[i])


plt.figure(figsize=(10, 6))
plt.title('How volatility affects the variance of terminal prices')
plt.xlabel('Volatility')
plt.ylabel('Variance of terminal prices')
plt.scatter(x_axis, y_axis);
